4 · 部署

三个脚本，彼此独立，可以分别在不同机器上跑。全部幂等：改完配置重跑即可，不会留下半成品。

把脚本传上去

scp -r docs/quality/scripts root@<服务器>:/root/ssh-spa     # 或者只 scp 你要用的那一个

第 1 步 · 生成密钥

在你自己安全的机器上跑，不要在服务器上生成：

bash 00-gen-keys.sh --hosts "server1=203.0.113.10 server2=198.51.100.7" --clients "laptop phone"

产出 spa-keys.env（服务端，含全部设备）+ clients/{laptop,phone}.env。详见 第 3 页。

第 2 步 · 服务端

scp spa-keys.env root@<服务器>:/root/
ssh root@<服务器> 'bash /root/ssh-spa/10-server.sh --keys /root/spa-keys.env'

按顺序做了五件事：

  1. 装 fwknop-server（Debian 包名，客户端才是 fwknop-client）
  2. 渲染 /etc/fwknop/fwknopd.conf：按默认路由网卡写 PCAP_INTF，嗅探端口取 SPA_PORT，跳转规则放在 INPUT 第 1 位，防重放三项全开。包自带版本备份为 fwknopd.conf.orig
  3. 渲染 /etc/fwknop/access.conf（0600）：从密钥包逐个 stanza 展开，并自检 stanza 数 == 密钥组数
  4. 启动 fwknop-server：起不来就打日志退出，此时防火墙还没动；起来了还要确认 FWKNOP_INPUT 跳转真的插上了
  5. 防火墙：ufw allow 62201/udp，22 保持原样；改完 ufw 后自动重启一次 fwknopd（见 第 6 页）
渲染出来的 fwknopd.conf
PCAP_INTF                eth0;
PCAP_FILTER              udp port 62201;

FLUSH_IPT_AT_INIT        Y;
FLUSH_IPT_AT_EXIT        Y;
ENABLE_IPT_COMMENT_CHECK Y;

# 防重放：时间戳 + 摘要缓存（进程重启后仍认得旧包）
ENABLE_SPA_PACKET_AGING  Y;
MAX_SPA_PACKET_AGE       120;
ENABLE_DIGEST_PERSISTENCE Y;

# 跳转规则插在 INPUT 第 1 位，因此在 ufw 的链之前求值
IPT_INPUT_ACCESS         ACCEPT, filter, INPUT, 1, FWKNOP_INPUT, 1;
ENABLE_IPT_FORWARDING    N;
ENABLE_IPT_SNAT          N;
ENABLE_IPT_OUTPUT        N;
ENABLE_IPT_LOCAL_NAT     N;

SYSLOG_ID                fwknopd;

第 3 步 · 客户端

每台机器各跑一次，只给它自己那一份密钥：

scp clients/laptop.env <你的机器>:~/
ssh <你的机器> 'bash 20-client.sh --keys ~/laptop.env'

做了三件事：装 fwknop；写 ~/.fwknoprc（0600）；往 ~/.ssh/config 追加一个托管块。

托管块是幂等的。带 # BEGIN ssh-spa … # END ssh-spa 标记，重跑只原地替换并清掉多余空行，不会越堆越多；--uninstall 摘掉之后与原文件逐字节一致。
参数作用
--keys FILE必填。自己那一份密钥包
--name NAME覆盖包里的 CLIENT_NAME
--no-ssh只写 ~/.fwknoprc，不碰 ~/.ssh/config
--status显示当前配置
--uninstall还原 ~/.fwknoprc，删掉托管块

装包路径按顺序探测：brew → apt（fwknop-client）→ pacman。缺 nc 只警告不中止，因为 ProxyCommand 的最后一步要它（Debian/Ubuntu：apt install netcat-openbsd）。

脚本全文
#!/usr/bin/env bash
# 20-client.sh —— 客户端：装 fwknop、渲染 ~/.fwknoprc、往 ~/.ssh/config 里塞 ProxyCommand。
#   每台机器跑一次，只给它自己那一份密钥（clients/<name>.env）。
set -euo pipefail
SELF_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
# shellcheck source=lib.sh
. "$SELF_DIR/lib.sh"

KEYS=""
NAME=""
SSH_CONFIG=1
MODE=deploy
BEGIN="# BEGIN ssh-spa (managed by 20-client.sh)"
END="# END ssh-spa"

usage() {
  cat <<'EOF'
用法：bash 20-client.sh --keys FILE [选项]

  --keys FILE     自己那一份密钥包（00-gen-keys.sh 生成的 clients/<名字>.env）
  --name NAME     覆盖包里的 CLIENT_NAME
  --no-ssh        只写 ~/.fwknoprc，不碰 ~/.ssh/config
  --status        显示当前配置
  --uninstall     还原 ~/.fwknoprc，删掉 ~/.ssh/config 里的托管块
  -h, --help      显示帮助
EOF
}

while [ $# -gt 0 ]; do
  case "$1" in
    --keys) shift; KEYS="${1:?--keys 需要参数}" ;;
    --keys=*) KEYS="${1#*=}" ;;
    --name) shift; NAME="${1:?--name 需要参数}" ;;
    --name=*) NAME="${1#*=}" ;;
    --no-ssh) SSH_CONFIG=0 ;;
    --status) MODE=status ;;
    --uninstall) MODE=uninstall ;;
    -h|--help) usage; exit 0 ;;
    *) die "未知参数：$1（--help 看用法）" ;;
  esac
  shift
done

FWKRC="${FWKRC:-$HOME/.fwknoprc}"
SSHCFG="${SSHCFG:-$HOME/.ssh/config}"

strip_block() {  # 从 ~/.ssh/config 里摘掉托管块，并清掉末尾多余空行（否则重跑会越积越多）
  [ -f "$SSHCFG" ] || return 0
  local tmp="$SSHCFG.spa.tmp"
  awk -v b="$BEGIN" -v e="$END" '
    $0 == b { skip = 1; next }
    $0 == e { skip = 0; next }
    !skip { print }
  ' "$SSHCFG" | sed -e :a -e '/^\n*$/{$d;N;ba' -e '}' > "$tmp" && mv "$tmp" "$SSHCFG"
}

if [ "$MODE" = status ]; then
  printf '%s : %s\n' "$FWKRC" "$([ -f "$FWKRC" ] && echo 存在 || echo 不存在)"
  [ -f "$FWKRC" ] && grep -E '^\[' "$FWKRC" | tr -d '[]' | sed 's/^/  stanza: /'
  grep -qF "$BEGIN" "$SSHCFG" 2>/dev/null && echo "$SSHCFG : 托管块存在" || echo "$SSHCFG : 无托管块"
  exit 0
fi

[ -n "$KEYS" ] || { usage; die "--keys 是必填的"; }

if [ "$MODE" = uninstall ]; then
  strip_block
  if [ -f "$FWKRC" ] && grep -qF '由 20-client.sh 生成' "$FWKRC"; then
    rm -f "$FWKRC"
    log "已删除 $FWKRC"
  fi
  log "卸载完成。密钥包 $KEYS 自己留着或删掉都行"
  exit 0
fi

# ---------------------------------------------------------------- 安装

log "1/3 安装 fwknop"
if ! command -v fwknop >/dev/null 2>&1; then
  if command -v brew >/dev/null 2>&1; then
    brew install fwknop >/dev/null
  elif command -v apt-get >/dev/null 2>&1; then
    sudo env DEBIAN_FRONTEND=noninteractive apt-get install -y -qq fwknop-client >/dev/null
  elif command -v pacman >/dev/null 2>&1; then
    sudo pacman -S --noconfirm fwknop >/dev/null
  else
    die "没找到 fwknop，手工装一个（Debian/Ubuntu: apt install fwknop-client）"
  fi
fi
command -v fwknop >/dev/null 2>&1 || die "fwknop 装完仍找不到"
command -v nc >/dev/null 2>&1 || warn "没有 nc，ProxyCommand 里的最后一步会失败（Debian/Ubuntu: apt install netcat-openbsd）"

# ---------------------------------------------------------------- 渲染 ~/.fwknoprc

load_env "$KEYS"
[ -n "$NAME" ] && { valid_name "$NAME" || die "客户端名不合法：$NAME"; CLIENT_NAME="$NAME"; }
: "${SPA_PORT:=62201}" "${SSH_PORT:=22}" "${ACCESS_TIMEOUT:=120}"
[ -n "${HOSTS:-}" ] || die "密钥包里没有 HOSTS，重新生成或手工加一行 HOSTS=\"名字=地址\""

# 没指定名字、文件里也没有 CLIENT_NAME，就走 SHARED（spa_keys_for 自己会回退）
PAIR="$(spa_keys_for "$KEYS" "${CLIENT_NAME:-}")"
WHO="${CLIENT_NAME:-shared（共用密钥组）}"

log "2/3 渲染 $FWKRC（客户端 $WHO）"
umask 077
# fwknop 装完会丢一个全注释的样例，只有那类文件值得备份；我们生成的版本不用备
[ -f "$FWKRC" ] && ! grep -qF '由 20-client.sh 生成' "$FWKRC" && cp -a "$FWKRC" "$FWKRC.bak"
{
  printf '# 由 20-client.sh 生成，勿手改。重跑脚本即可更新。\n'
  printf '# 客户端 %s（服务端在 access.conf 里按这把密钥逐个 stanza 试）\n\n' "$WHO"
  printf '[default]\n'
  printf 'SPA_SERVER_PROTO  udp\n'
  printf 'USE_HMAC          Y\n'
  printf 'HMAC_DIGEST_TYPE  SHA512\n'
  printf 'ALLOW_IP          resolve\n'
  printf 'ACCESS            tcp/%s\n' "$SSH_PORT"
  printf 'KEY_BASE64        %s\n' "${PAIR%%:*}"
  printf 'HMAC_KEY_BASE64   %s\n' "${PAIR##*:}"
  for h in $HOSTS; do
    printf '\n[%s]\n' "${h%%=*}"
    printf 'SPA_SERVER        %s\n' "${h#*=}"
    printf 'SPA_SERVER_PORT   %s\n' "$SPA_PORT"
  done
} > "$FWKRC"
chmod 600 "$FWKRC"

# 自检：写完的文件要含每个主机 stanza，且密钥对只有一个冒号
for h in $HOSTS; do
  grep -qF "[${h%%=*}]" "$FWKRC" || die "自检失败：$FWKRC 里少了 stanza ${h%%=*}"
done
case "$PAIR" in
  *:*) ;;
  *) die "自检失败：密钥对格式不对" ;;
esac

# ---------------------------------------------------------------- ~/.ssh/config

if [ "$SSH_CONFIG" = 1 ]; then
  log "3/3 写 $SSHCFG 托管块"
  mkdir -p "$(dirname "$SSHCFG")"; chmod 700 "$(dirname "$SSHCFG")"
  strip_block
  # 备份摘掉托管块之后的版本：备份里不含我们塞的东西
  [ -s "$SSHCFG" ] && cp -a "$SSHCFG" "$SSHCFG.bak"
  {
    cat "$SSHCFG"
    printf '\n%s\n' "$BEGIN"
    for h in $HOSTS; do
      printf 'Host %s\n' "${h%%=*}"
      printf '  HostName %s\n' "${h#*=}"
      # 敲门失败也要往下走：新机器上 fwknop 还没装，nc 会直接连上（那时 22 还开着）
      printf '  ProxyCommand fwknop -n %s >/dev/null 2>&1; sleep 2; nc %%h %%p\n' "${h%%=*}"
    done
    printf '%s\n' "$END"
  } > "$SSHCFG.tmp"
  mv "$SSHCFG.tmp" "$SSHCFG"
  chmod 600 "$SSHCFG"
fi

cat <<EOF

客户端就绪。测一下：
  fwknop -n <主机别名>      # 手动敲一次，服务端 iptables -S FWKNOP_INPUT 会多一条规则
  ssh <主机别名>             # ProxyCommand 里已经带了敲门，感知不到

被自己锁在门外时的急救（本机另开一个终端，不要关掉现在这个）：
  sudo systemctl stop fwknop-server && sudo iptables -D INPUT -j FWKNOP_INPUT
  sudo ufw allow 22/tcp
EOF

第 4 步 · 收口

顺序不能反。先在当前这个 SSH 会话里另开一个连接，确认能进得来，再收口。
bash /root/ssh-spa/10-server.sh --keys /root/spa-keys.env --close-ssh

它会删除公网 22/tcp 规则（ufw limit 和 ufw allow 两种都试），然后重启 fwknopd 恢复跳转规则。

锁门急救

万一还是锁了——云厂商控制台的 VNC / 串口，或者你之前装了 Tailscale（见 第 6 页）：

sudo systemctl stop fwknop-server
sudo iptables -D INPUT -j FWKNOP_INPUT
sudo ufw allow 22/tcp

或者直接让脚本还原（保留密钥包）：

sudo bash 10-server.sh --uninstall     # 还原 fwknopd.conf，恢复公网 SSH 规则
sudo bash 10-server.sh --purge         # 上面这些，再删掉 access.conf 与密钥包

服务端参数

参数作用
--keys FILE密钥包路径，默认 ./spa-keys.env
--close-ssh删掉公网 SSH 规则。确认客户端能进门之后再加
--status只看状态，不改任何东西
--uninstall停服务、还原 fwknopd.conf、恢复公网 SSH 规则
--purge上面这些，再删掉 access.conf 和密钥包
脚本全文
#!/usr/bin/env bash
# 10-server.sh —— 服务端：装 fwknopd、渲染 access.conf、收掉公网 SSH 规则。
#   幂等：重跑只补齐缺失部分。--close-ssh 是单独一步，避免把管理员锁在门外。
set -euo pipefail
SELF_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
# shellcheck source=lib.sh
. "$SELF_DIR/lib.sh"
need_root

FWK_DIR="${FWK_DIR:-/etc/fwknop}"
FWD_CONF="$FWK_DIR/fwknopd.conf"
ACCESS_CONF="$FWK_DIR/access.conf"
FWD_ORIG="$FWK_DIR/fwknopd.conf.orig"
KEYS="spa-keys.env"
CLOSE_SSH=0
PURGE=0
MODE=deploy

usage() {
  cat <<'EOF'
用法：sudo bash 10-server.sh [选项]

  --keys FILE       00-gen-keys.sh 生成的密钥包（默认 ./spa-keys.env）
  --close-ssh       删掉公网 SSH 防火墙规则。确认能从客户端敲门进来之后再加这个参数
  --status          只看状态，不改任何东西
  --uninstall       停 fwknopd、还原 fwknopd.conf、恢复公网 SSH 规则（保留密钥包）
  --purge           --uninstall + 删除 access.conf 和密钥包
  -h, --help        显示帮助
EOF
}

while [ $# -gt 0 ]; do
  case "$1" in
    --keys) shift; KEYS="${1:?--keys 需要参数}" ;;
    --keys=*) KEYS="${1#*=}" ;;
    --close-ssh) CLOSE_SSH=1 ;;
    --status) MODE=status ;;
    --uninstall) MODE=uninstall ;;
    --purge) MODE=uninstall; PURGE=1 ;;
    -h|--help) usage; exit 0 ;;
    *) die "未知参数：$1（--help 看用法）" ;;
  esac
  shift
done

# ---------------------------------------------------------------- 状态

show_status() {
  printf '包      : %s\n' "$KEYS"
  if command -v fwknopd >/dev/null 2>&1; then
    printf 'fwknopd : %s\n' "$(systemctl is-active fwknop-server 2>/dev/null || echo 未安装)"
    printf '跳转规则: %s\n' "$(iptables -S INPUT 2>/dev/null | grep -q FWKNOP_INPUT && echo '在 INPUT 第 1 位（正常）' || echo '缺失 —— 跑 systemctl restart fwknop-server，或检查 ufw 是否重载过')"
    printf '当前放行: %s\n' "$(iptables -S FWKNOP_INPUT 2>/dev/null | grep -c '^-A' || true) 条临时规则"
  else
    printf 'fwknopd : 未安装\n'
  fi
  if command -v ufw >/dev/null 2>&1; then
    printf 'ufw     : %s\n' "$(ufw status | head -1)"
    ufw status | grep -E "22/tcp|62201/udp" | sed 's/^/          /' || true
  fi
}

if [ "$MODE" = status ]; then show_status; exit 0; fi

# ---------------------------------------------------------------- 卸载

if [ "$MODE" = uninstall ]; then
  systemctl disable --now fwknop-server >/dev/null 2>&1 || true
  [ -f "$FWD_ORIG" ] && { mv "$FWD_ORIG" "$FWD_CONF"; log "已还原 fwknopd.conf"; }
  [ "$PURGE" = 1 ] && { rm -f "$ACCESS_CONF" "$KEYS"; log "已删除 access.conf 与密钥包"; }
  if command -v ufw >/dev/null 2>&1; then
    ufw --force delete allow 62201/udp >/dev/null 2>&1 || true
    ufw limit 22/tcp >/dev/null 2>&1 || true
    log "已恢复公网 SSH 规则（ufw limit 22/tcp）"
  fi
  exit 0
fi

# ---------------------------------------------------------------- 部署

need_cmd openssl
command -v apt-get >/dev/null 2>&1 || die "本脚本按 Debian/Ubuntu 写的；其他发行版请照 README 手工改"

log "1/5 安装 fwknop-server"
if ! command -v fwknopd >/dev/null 2>&1; then
  DEBIAN_FRONTEND=noninteractive apt-get update -qq
  DEBIAN_FRONTEND=noninteractive apt-get install -y -qq fwknop-server >/dev/null
fi
command -v fwknopd >/dev/null 2>&1 || die "fwknopd 装完仍找不到，检查 apt 输出"

load_env "$KEYS"
: "${SPA_PORT:=62201}" "${SSH_PORT:=22}" "${ACCESS_TIMEOUT:=120}" "${MAX_FW_TIMEOUT:=300}"

IFACE="$(ip -4 route show default 2>/dev/null | awk '{print $5; exit}')"
[ -n "$IFACE" ] || die "识别不到默认网卡，PCAP_INTF 只能靠 fwknopd 自己猜，指定网卡再跑"

# ---- 渲染 fwknopd.conf：只写我们要管的几项，其余用包自带默认值 ----
log "2/5 渲染 $FWD_CONF（网卡 $IFACE，SPA 端口 $SPA_PORT）"
mkdir -p "$FWK_DIR"
[ -f "$FWD_ORIG" ] || { [ -f "$FWD_CONF" ] && cp -a "$FWD_CONF" "$FWD_ORIG"; }
cat > "$FWD_CONF" <<EOF
# 由 10-server.sh 生成，勿手改。包自带版本备份在 $FWD_ORIG
PCAP_INTF                $IFACE;
PCAP_FILTER              udp port $SPA_PORT;

FLUSH_IPT_AT_INIT        Y;
FLUSH_IPT_AT_EXIT        Y;
ENABLE_IPT_COMMENT_CHECK Y;

# 防重放：时间戳 + 摘要缓存（进程重启后仍认得旧包）
ENABLE_SPA_PACKET_AGING  Y;
MAX_SPA_PACKET_AGE       120;
ENABLE_DIGEST_PERSISTENCE Y;

# 跳转规则插在 INPUT 第 1 位，因此在 ufw 的链之前求值：SPA 开的口子不会被 ufw 挡掉
IPT_INPUT_ACCESS         ACCEPT, filter, INPUT, 1, FWKNOP_INPUT, 1;
ENABLE_IPT_FORWARDING    N;
ENABLE_IPT_SNAT          N;
ENABLE_IPT_OUTPUT        N;
ENABLE_IPT_LOCAL_NAT     N;

SYSLOG_ID                fwknopd;
EOF
chmod 644 "$FWD_CONF"

# ---- 渲染 access.conf：一个客户端一个 stanza ----
# fwknopd 会依次尝试每个 stanza，用各自的密钥解密；只有对上的那个能开规则
# （server/incoming_spa.c: incoming_spa() 里的 while(acc) 循环）。所以所有 stanza
# 都可以写 SOURCE ANY，靠密钥区分客户端。
log "3/5 渲染 $ACCESS_CONF"
umask 077
render_stanza() {
  printf 'SOURCE ANY;\n'
  printf 'OPEN_PORTS tcp/%s;\n' "$SSH_PORT"
  printf 'FW_ACCESS_TIMEOUT %s;\n' "$ACCESS_TIMEOUT"
  printf 'MAX_FW_TIMEOUT %s;\n' "$MAX_FW_TIMEOUT"
  printf 'KEY_BASE64 %s;\n' "${1%%:*}"
  printf 'HMAC_KEY_BASE64 %s;\n' "${1##*:}"
  printf '\n'
}

{
  printf '# 由 10-server.sh 从 %s 渲染，勿手改。\n' "$KEYS"
  printf '# 每个 stanza 一台客户端：删掉一行就是吊销那台设备。\n\n'
  n=0
  if [ -n "${SHARED:-}" ]; then
    n=$((n+1))
    printf '# --- shared（共用密钥组）---\n'
    render_stanza "$SHARED"
  fi
  for entry in ${CLIENTS:-}; do
    n=$((n+1))
    printf '# --- %s ---\n' "${entry%%=*}"
    render_stanza "${entry#*=}"
  done
  [ "$n" -gt 0 ] || die "密钥包里没有 SHARED 也没有 CLIENTS"
} > "$ACCESS_CONF"
chmod 600 "$ACCESS_CONF"

# 自检：渲染出来的 stanza 数必须等于密钥包里的密钥组数
want=$(host_count "${CLIENTS:-}")
[ -n "${SHARED:-}" ] && want=$((want+1))
got="$(grep -c '^KEY_BASE64 ' "$ACCESS_CONF" || true)"
[ "$got" = "$want" ] || die "自检失败：access.conf 渲染出 $got 个 stanza，期望 $want 个"

# ---- 起服务 ----
log "4/5 启动 fwknopd"
systemctl enable fwknop-server >/dev/null 2>&1 || true
systemctl restart fwknop-server
sleep 1
systemctl is-active --quiet fwknop-server || {
  journalctl -u fwknop-server -n 20 --no-pager >&2
  die "fwknopd 起不来（多半是配置语法错）。SSH 规则还没动，可以直接看上面的日志"
}
iptables -S INPUT | grep -q FWKNOP_INPUT || die "fwknopd 在跑但没有创建 FWKNOP_INPUT 跳转，检查 $FWD_CONF 的 PCAP_INTF"

# ---- 防火墙 ----
log "5/5 防火墙"
if command -v ufw >/dev/null 2>&1; then
  ufw allow "$SPA_PORT/udp" >/dev/null
  log "已放行 $SPA_PORT/udp"
  if [ "$CLOSE_SSH" = 1 ]; then
    ufw --force delete limit "$SSH_PORT/tcp" >/dev/null 2>&1 || true
    ufw --force delete allow "$SSH_PORT/tcp" >/dev/null 2>&1 || true
    log "已删除公网 $SSH_PORT/tcp 规则"
  else
    log "保持公网 $SSH_PORT/tcp 规则（确认客户端能进门后跑 --close-ssh）"
  fi
  # ufw 的 enable/reload/reset 会重建 INPUT 链，把 fwknopd 的跳转冲掉 → 重启让它插回去
  systemctl restart fwknop-server
  sleep 1
  iptables -S INPUT | grep -q FWKNOP_INPUT || warn "重启后跳转规则仍不在，确认 iptables 是不是 nft 后端"
else
  warn "没装 ufw，跳过防火墙改动。自己确认 $SPA_PORT/udp 通，且 $SSH_PORT/tcp 不对公网开放"
fi

cat <<EOF

服务端就绪。下一步：
  1) 在客户端跑 20-client.sh 配好 ~/.fwknoprc
  2) 先在当前这个 SSH 会话里另开一个连接试：ssh <别名>
     确认能进来，再跑：bash 10-server.sh --keys $KEYS --close-ssh
  3) 从外网验一遍：nmap -Pn -p$SSH_PORT <服务器IP>   → 必须显示 filtered

共用函数

三个脚本共用 lib.sh（log/warn/die、need_root、load_env、spa_keys_for、valid_name、host_count、gen_key）。改密钥包格式只需要动 spa_keys_for 一个函数。

脚本全文
#!/usr/bin/env bash
# ssh-spa 共用函数。不要单独运行，被 00/10/20-*.sh source。

log()  { printf '\033[32m[+]\033[0m %s\n' "$*"; }
warn() { printf '\033[33m[!]\033[0m %s\n' "$*" >&2; }
die()  { printf '\033[31m[x]\033[0m %s\n' "$*" >&2; exit 1; }

need_root() { [ "${SKIP_ROOT:-0}" = 1 ] && return 0; [ "$(id -u)" = 0 ] || die "需要 root：sudo bash $0"; }
need_cmd() { command -v "$1" >/dev/null 2>&1 || die "缺少命令 $1"; }

# 读取 KEY=VALUE 环境文件（只做赋值，不执行其中的命令）
load_env() {
  local f="$1"
  [ -f "$f" ] || die "找不到密钥包：$f"
  set -a; . "$f"; set +a
}

# 名字只允许字母数字 . _ -，因为它会进 grep -E 和 fwknop 的 stanza 名
valid_name() { case "${1:-}" in "" | *[!A-Za-z0-9._-]*) return 1 ;; *) return 0 ;; esac; }

# 取密钥对 "<KEY_BASE64>:<HMAC_KEY_BASE64>"：
# 传了名字（或文件里有 CLIENT_NAME）就找该客户端，否则回退到 SHARED
# 每客户端的密钥放在一行 CLIENTS="name=k:h other=k:h" 里，
# 不能写成 CLIENT name=... —— 那种行 shell source 不了。
spa_keys_for() {
  local file="$1" name="${2:-}" line entry
  [ -n "$name" ] || name="${CLIENT_NAME:-}"
  if [ -n "$name" ]; then
    line="$(grep -E '^CLIENTS=' "$file" | head -n1 || true)"
    [ -n "$line" ] || die "密钥包里没有 CLIENTS"
    # 值在文件里带引号，base64 字符集不含引号，直接去掉
    for entry in $(printf '%s' "${line#*=}" | tr -d '"'); do
      case "$entry" in
        "$name"=*) printf '%s\n' "${entry#*=}"; return 0 ;;
      esac
    done
    die "密钥包里没有客户端 $name（重新生成时把它写进 --clients）"
  fi
  line="$(grep -E '^SHARED=' "$file" | head -n1 || true)"
  [ -n "$line" ] || die "密钥包里既没有客户端 $name 也没有 SHARED"
  printf '%s\n' "${line#*=}"
}

# 数一个 "a=1 b=2" 形式的值里有几项
host_count() { local v="${1:-}"; [ -n "$v" ] || { echo 0; return; }; set -- $v; echo $#; }

# 32 字节随机 base64，等价于 fwknop --key-gen 的输出格式
gen_key() { openssl rand -base64 32 | tr -d '\n'; }
